from pathlib import Path
from zipfile import ZipFile
from IPython.display import FileLink, display

zip_path = Path.cwd() / f"{RUN_NAME}_results.zip"

result_files = list(config.RESULTS_DIR.glob(f"{RUN_NAME}*"))
checkpoint_file = CHECKPOINT_DIR / f"{RUN_NAME}.pt"
if checkpoint_file.exists():
    result_files.append(checkpoint_file)

with ZipFile(zip_path, "w") as archive:
    for path in result_files:
        archive.write(path, arcname=path.name)

print("ZIP contains:", [path.name for path in result_files])
display(FileLink(str(zip_path)))
